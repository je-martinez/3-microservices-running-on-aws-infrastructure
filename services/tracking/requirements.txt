# building today. Bump deliberately, never implicitly.

# runtime image carries no test/lint toolchain. Runtime dependencies go THERE.
-r requirements-runtime.txt

ruff==0.9.2
# Required by `fastapi.testclient.TestClient`, which drives the REST suite against
# the real app in-process (ASGI transport, no socket). Test-only — nothing under
# src/ imports it.
httpx==0.28.1
# Spec generation only (`scripts/generate_openapi.py`), never imported at runtime
# — FastAPI builds the OpenAPI document as a dict on its own; this only serializes

# grpcio-tools below: the committed `openapi.yaml` is what consumers read, so the
# runtime image needs no YAML writer.

# Codegen only (`scripts/generate_grpc_stubs.py`), never imported at runtime —
# the stubs it produces are committed, so the image needs no protoc. Kept at the
# SAME version as grpcio: grpc_tools emits a version guard into the generated
# _pb2_grpc module that raises when the installed grpcio is older.
grpcio-tools==1.83.0
# In-process Redis double for the cache suite. Chosen over a real Redis on purpose:
# the DB suite must hit real MySQL (a mock cannot catch a schema or driver bug — see
# tests/conftest.py), but a cache has NO schema and NO dialect, so a real server buys
# nothing and costs a hard dependency on Floci's valkey container being up for a suite
# that otherwise only needs MySQL. fakeredis implements the exact `redis` client API
# this code calls (get/setex/delete/sadd/smembers/ttl/expire), including TTL
# semantics, so the code under test is the production code path.
fakeredis==2.26.2
